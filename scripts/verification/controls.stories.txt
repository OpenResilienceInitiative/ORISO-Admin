const ThemedFeedbackContent = () => {
    const { token } = theme.useToken();
    return <span data-feedback-token={token.colorPrimary} style={{ color: token.colorPrimary }}>Contextual feedback</span>;
};
const FeedbackControl = () => {
    const { message } = useAdminFeedback();
    const { token } = theme.useToken();
    return <button data-feedback-source={token.colorPrimary} onClick={() => message.success({ content: <ThemedFeedbackContent />, duration: 0 })}>Show themed feedback</button>;
};
export const ContextualFeedback = {
    render: () => <FeedbackControl />,
    play: async ({ canvasElement }) => {
        const button = within(canvasElement).getByRole('button', { name: 'Show themed feedback' });
        await userEvent.click(button);
        const content = await within(document.body).findByText('Contextual feedback');
        expect(content.getAttribute('data-feedback-token')).toBe(button.getAttribute('data-feedback-source'));
        const expectedColor = document.createElement('span');
        expectedColor.style.color = button.getAttribute('data-feedback-source');
        document.body.append(expectedColor);
        expect(getComputedStyle(content).color).toBe(getComputedStyle(expectedColor).color);
        expectedColor.remove();
        expect(content.closest('.ant-message')).not.toBeNull();
    },
};
